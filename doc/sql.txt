** create table counter **
CREATE TABLE counter (
    counter_id INTEGER PRIMARY KEY AUTOINCREMENT,
    position INTEGER NOT NULL
);

**create sevice**
CREATE TABLE services (
    service_id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(50) NOT NULL,
    description varchar(255)
);


** table n-n services and counter**

CREATE TABLE counter_services (
    counter_id INTEGER NOT NULL,
    service_id INTEGER NOT NULL,

    PRIMARY KEY (counter_id, service_id),

    FOREIGN KEY (counter_id)
        REFERENCES counter(counter_id)
        ,

    FOREIGN KEY (service_id)
        REFERENCES services(service_id)
        
);

**create tickets**
CREATE TABLE tickets (
    ticket_id INTEGER PRIMARY KEY AUTOINCREMENT,
    ticket_cod VARCHAR(20) NOT NULL UNIQUE,
    service_id INTEGER NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'WAITING',
    created_time TIME NOT NULL DEFAULT (TIME('now', 'localtime')),

    FOREIGN KEY (service_id)
        REFERENCES services(service_id)
);
